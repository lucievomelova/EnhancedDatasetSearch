File containing methods related specifically to the NKOD portal.

There will be a database containing info about all datasets. Every day, the new datove_sady and distribuce csvs will be downloaded and if there are changes detected in some datasets at NKOD, their info will be deleted from DB and then added again.

141    def _transform_datasets_raw(self) -> None:
142        """Apply initial transformations on the raw dataset."""
143        today = datetime.today().date()
144        mod_datetime = None
145        if os.path.exists(self._data_config["datasets_transformed_path"]):
146            mod_time = os.path.getmtime(self._data_config["datasets_transformed_path"])
147            mod_datetime = datetime.fromtimestamp(mod_time)
148        #  check if datasets_raw_transformed file is up to date - if it is, we can load the data from there
149        if mod_datetime is not None and mod_datetime.date() == today:
150            logger.info("File containing datasets_raw_transformed exists and is up to date - loading.")
151            self.datasets_raw_transformed = pd.read_csv(
152                self._data_config["datasets_transformed_path"], sep=",", converters={col: pd.eval for col in self.list_columns})
153        else:
154            # transform data from datasets_raw to obtained datasets_raw_transformed
155            self.datasets_raw_transformed = self.datasets_raw.rename(columns=self._column_mapping)
156            self.datasets_raw_transformed = merge_keywords_and_themes_rows(self.datasets_raw_transformed)
157            self.datasets_raw_transformed = drop_irrelevant_columns(self.datasets_raw_transformed,
158                                                                    self._data_processing_config["irrelevant_columns"])
159
160            add_metadata_to_datasets_from_sparql(self.config, self.datasets_raw_transformed)
161
162            # store full description hash before any processing is applied so that later we are able to detect changes
163            # in datasets' metadata
164            self.datasets_raw_transformed["full_description"] = self.datasets_raw_transformed.apply(
165                lambda r: f"{r["title"]}\n{r["description"]}\n({r["provider"]} | "
166                          f"{", ".join(sorted(r["keywords"]))} | "
167                          f"{", ".join(sorted(r["themes"]))} | "
168                          f"{", ".join(sorted(r["spatial_coverage"]))} | "
169                          f"{", ".join(sorted(r["temporal_coverage"]))})", axis=1
170            )
171            self.datasets_raw_transformed["full_description_hash"] = self.datasets_raw_transformed.apply(
172                lambda r: hashlib.md5(r["full_description"].encode("utf-8")).hexdigest(), axis=1
173            )
174            # add category column, now empty for each dataset
175            self.datasets_raw_transformed["categories"] = [[] for _ in range(len(self.datasets_raw_transformed))]
176            clean_metadata(self.datasets_raw_transformed,
177                           self.client,
178                           self._data_processing_config["categories"],
179                           self._state_dir)
180
181            self.datasets_raw_transformed.to_csv(self._data_config["datasets_transformed_path"], index=False, header=True)
182
183        # metadata cleaned -> find all unique keywords and themes, which will be used for metadata enrichment
184        self._all_keywords_raw = set(self.datasets_raw_transformed["keywords"].explode().dropna().unique())
185        logger.info(f"Number of unique keywords in transformed datasets_raw: {len(self._all_keywords_raw)}")

202    def get_new_datasets(self) -> tuple[pd.DataFrame, list]:
203        """Get a dataframe of new or updated datasets and update self.datasets with the new data."""
204        # find new or updated datasets by comparing new and old datasets if old datasets exist
205        removed_urls = None  # for tracking which urls were present before but are not present now
206        if not self.datasets.empty:
207            # merge by url and full description to find all changes
208            merged_df = pd.merge(
209                self.datasets_raw_transformed, self.datasets[["url", "full_description_hash"]],
210                on=["url", "full_description_hash"], how='outer', indicator=True
211            )
212
213            # get urls of removed and updated datasets, so we can remove them
214            removed_urls = merged_df[merged_df['_merge'] == 'right_only']["url"].tolist()
215
216            # new OR updated datasets, but we remove updated datasets and process them again, so we call them all new
217            new_datasets = merged_df[merged_df['_merge'] == 'left_only'][self.datasets_raw_transformed.columns]
218            new_datasets = new_datasets.drop(columns="full_description")
219
220            logger.info(f"Number of new or updated datasets: {new_datasets.shape[0]}.")
221        else:  # otherwise all datasets are new
222            logger.info(f"Old file not found. Adding all datasets to DB ({self.datasets_raw_transformed.shape[0]} datasets).")
223            new_datasets = self.datasets_raw_transformed
224
225        # update datasets - process new datasets and add them to the existing datasets dataframe
226        if not self.datasets.empty:
227            if removed_urls:   # remove deleted datasets from self.datasets
228                self.datasets = self.datasets[~self.datasets['url'].isin(removed_urls)]
229                logger.info(f"Removed {len(removed_urls)} datasets from DB.")
230            # find which rows are already in the datasets based on url - don't add them again
231            merged_df = pd.merge(self.datasets["url"], new_datasets, on="url", how='outer', indicator=True)
232            new_datasets = merged_df.query("_merge == 'right_only'").drop('_merge', axis=1).reset_index(drop=True)
233        if removed_urls is None:
234            removed_urls = []
235        return new_datasets, removed_urls

237    async def enrich_new_datasets_metadata(self, new_datasets: pd.DataFrame) -> None:
238        """Enrich metadata of new_datasets and update self.datasets with the enriched data."""
239        # file for storing intermediate results - it must be csv, because json doesn't have an append option
240        # but later we want to use json because of faster loading time
241        tmp_file_name = self._data_config["datasets_path"].replace(".json", f"_tmp.csv")
242        if not self.datasets.empty:  # if datasets is not empty, put current state of it in tmp file
243            self.datasets.to_csv(tmp_file_name, index=False, header=True)
244        # do the updates in chunks -> in case of script failure we can resume from the last chunk
245        chunks = split_dataframe(new_datasets, batch_size=32)
246        logger.info(f"Extending dataset metadata.")
247        for index, chunk in enumerate(chunks, start=1):
248            if chunk.empty:
249                break
250            logger.info(f"Processing chunk {index}/{len(chunks)}.")
251            new_rows = await self._create_metadata_for_chunk(chunk)
252            new_rows_df = pd.DataFrame(new_rows)
253
254            # store / append new results
255            if not os.path.exists(tmp_file_name):
256                logger.info("Creating csv file for datasets.")
257                new_rows_df.to_csv(tmp_file_name, index=False, header=True)
258            else:
259                new_rows_df.to_csv(tmp_file_name, index=False, header=False, mode="a")
260
261        # load the csv that we were gradually writing to and save it as json, so that it can be
262        # loaded faster in subsequent loads, because we don't have to use pd converters for list columns
263        self.datasets = pd.read_csv(tmp_file_name, sep=",", converters={col: pd.eval for col in self.list_columns})
264        if os.path.exists(tmp_file_name):  # remove temporary file, it is not needed anymore
265            os.remove(tmp_file_name)
266
267        for col in self.list_columns:  # validate that each list column truly contains a list - otherwise set it as empty list
268            self.datasets[col] = self.datasets[col].apply(lambda x: x if isinstance(x, list) else [])
269
270        # metadata cleaning for the enhanced datasets
271        process_spatial_and_temporal_coverage(self.datasets)
272        clean_metadata(self.datasets, self.client, self._data_processing_config["categories"], self._state_dir)
273
274        # save after metadata cleaning as json
275        self.datasets.to_json(self._data_config["datasets_path"], orient="split", force_ascii=False)

277    async def update_datasets(self) -> tuple[pd.DataFrame, list]:
278        """Update self.datasets and find and return new or updated datasets and removed datasets.
279
280        Update datasets based on the last downloaded datasets_raw table. Remove datasets that are not present,
281        update existing datasets or add new datasets, then enrich new or existing datasets' metadata using an LLM.
282
283        Returns:
284            a tuple: (pd.DataFrame, list), where the dataframe contains new or updated datasets in hte same format as
285            self.dataset. The list contains a list of removed datasets URLs.
286        """
287        new_datasets, removed_urls = self.get_new_datasets()
288        await self.enrich_new_datasets_metadata(new_datasets)
289        new_datasets = self.datasets[self.datasets["url"].isin(new_datasets["url"])]
290        return new_datasets, removed_urls

292    async def _create_metadata_for_chunk(self, new_datasets:  pd.DataFrame):
293        """Create metadata dict for a chunk of new datasets asynchronously."""
294        new_rows = [self._enrich_datasets_metadata_async(row) for _, row in new_datasets.iterrows()]
295        return await asyncio.gather(*new_rows)

297    async def _enrich_datasets_metadata_async(self, row: Series) -> dict:
298        """Asynchronous wrapper for _enrich_datasets_metadata."""
299        # this method is needed for the async code to work properly
300        loop = asyncio.get_running_loop()
301        return await loop.run_in_executor(executor, self._enrich_datasets_metadata, row)

303    def _enrich_datasets_metadata(self, row: Series) -> dict:
304        """Enrich metadata of a dataset represented by the given row.
305
306        The metadata categories that are considered are: keywords, themes, categories, spatial_coverages and
307        temporal_coverage. If no (ior not enough) metadata is assigned to the dataset in a specific metadata category,
308        new metadata are generated by an LLM."""
309        categories = self._data_processing_config["categories"]
310        other_category = self._data_processing_config["other_category"]
311        generated_metadata = enrich_metadata(row, self.client, categories, other_category)
312
313        keywords = row["keywords"] if row["keywords"] is not None else []
314        keywords = list(set(keywords + generated_metadata["keywords"]))
315        themes = row["themes"] if row["themes"] is not None else []
316        categories = row["categories"] if row["categories"] is not None else []
317        categories = list(set(categories + generated_metadata["categories"]))
318        spatial_coverage = row["spatial_coverage"] if row["spatial_coverage"] is not None else []
319        spatial_coverage = list(set(spatial_coverage + generated_metadata["spatial_coverage"]))
320        temporal_coverage = row["temporal_coverage"] if row["temporal_coverage"] is not None else []
321        temporal_coverage = list(set(temporal_coverage + generated_metadata["temporal_coverage"]))
322
323        metadata = {
324            "title": row["title"],
325            "description": row["description"],
326            "url": row["url"],
327            "keywords": keywords,
328            "themes": themes,
329            "provider": row["provider"],
330            "categories": categories,
331            "spatial_coverage": spatial_coverage,
332            "temporal_coverage": temporal_coverage,
333            "full_description_hash": row["full_description_hash"]
334        }
335        return metadata

150        if mod_datetime is not None and mod_datetime.date() == today:
151            logger.info("File containing datasets_raw_transformed exists and is up to date - loading.")
152            self.datasets_raw_transformed = pd.read_csv(
153                self._data_config["datasets_transformed_path"], sep=",", converters={col: pd.eval for col in self.list_columns})
154        else:
155            # transform data from datasets_raw to obtained datasets_raw_transformed
156            self.datasets_raw_transformed = self.datasets_raw.rename(columns=self._column_mapping)
157            self.datasets_raw_transformed = merge_keywords_and_themes_rows(self.datasets_raw_transformed)
158            self.datasets_raw_transformed = drop_irrelevant_columns(self.datasets_raw_transformed,
159                                                                    self._data_processing_config["irrelevant_columns"])

161            add_metadata_to_datasets_from_sparql(self.config, self.datasets_raw_transformed)
162
163            # store full description hash before any processing is applied so that later we are able to detect changes
164            # in datasets' metadata
165            self.datasets_raw_transformed["full_description"] = self.datasets_raw_transformed.apply(
166                lambda r: f"{r["title"]}\n{r["description"]}\n({r["provider"]} | "
167                          f"{", ".join(sorted(r["keywords"]))} | "
168                          f"{", ".join(sorted(r["themes"]))} | "

178                           self.client,

180                           self._state_dir)

185        self._all_keywords_raw = set(self.datasets_raw_transformed["keywords"].explode().dropna().unique())

275        # save after metadata cleaning as json

304    def _enrich_datasets_metadata(self, row: Series) -> dict:
305        """Enrich metadata of a dataset represented by the given row.
306
307        The metadata categories that are considered are: keywords, themes, categories, spatial_coverages and
308        temporal_coverage. If no (ior not enough) metadata is assigned to the dataset in a specific metadata category,
309        new metadata are generated by an LLM."""
310        categories = self._data_processing_config["categories"]
311        other_category = self._data_processing_config["other_category"]
312        generated_metadata = enrich_metadata(row, self.client, categories, other_category)
313
314        keywords = row["keywords"] if row["keywords"] is not None else []
315        keywords = list(set(keywords + generated_metadata["keywords"]))
316        themes = row["themes"] if row["themes"] is not None else []
317        categories = row["categories"] if row["categories"] is not None else []
318        categories = list(set(categories + generated_metadata["categories"]))
319        spatial_coverage = row["spatial_coverage"] if row["spatial_coverage"] is not None else []
320        spatial_coverage = list(set(spatial_coverage + generated_metadata["spatial_coverage"]))
321        temporal_coverage = row["temporal_coverage"] if row["temporal_coverage"] is not None else []
322        temporal_coverage = list(set(temporal_coverage + generated_metadata["temporal_coverage"]))
323
324        metadata = {
325            "title": row["title"],
326            "description": row["description"],
327            "url": row["url"],
328            "keywords": keywords,
329            "themes": themes,
330            "provider": row["provider"],
331            "categories": categories,
332            "spatial_coverage": spatial_coverage,
333            "temporal_coverage": temporal_coverage,
334            "full_description_hash": row["full_description_hash"]
335        }
336        return metadata
